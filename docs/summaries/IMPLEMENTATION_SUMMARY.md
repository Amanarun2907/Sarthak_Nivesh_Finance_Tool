### Basic Usage

```python
# Initialize system

# Analyze any stock - demonstrates ALL 10 requirements!
result = system.analyze_stock(
    symbol="RELIANCE",
    user_query="Should I invest in this stock?",
    enable_all_agents=True

print(f"Reasoning: {result.reasoning}")
```

### Integration with Streamlit

```python
import streamlit as st
from core.unified_agentic_system import get_unified_system

@st.cache_resource
def init_system():
    return get_unified_system()

def main():
    st.title("🎯 Sarthak Nivesh - Agentic AI")
    
    system = init_system()
    symbol = st.text_input("Enter Stock Symbol")
    
    if st.button("Analyze"):
        with st.spinner("Analyzing..."):
            result = system.analyze_stock(symbol)
        
        st.success(f"**{result.action}** (Confidence: {result.confidence_score:.1f}%)")
        st.info(result.reasoning)
        
        with st.expander("View Agent Decisions"):
            for agent in result.agent_decisions:
                st.write(f"**{agent.agent_name}**: {agent.decision_type}")
```
