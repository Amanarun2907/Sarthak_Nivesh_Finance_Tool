// Example API call
const analyzeIPO = async (companyName, symbol) => {
    const response = await fetch('http://localhost:8000/api/ipo_intelligence/analyze', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            company_name: companyName,
            symbol: symbol,
            issue_price: 1000  // Optional
        })
    });
    
    const data = await response.json();
    
    // Display results
    console.log('Final Recommendation:', data.final_recommendation);
    console.log('Overall Score:', data.overall_score);
    console.log('Confidence:', data.overall_confidence * 100 + '%');
    
    // Individual agent scores
    console.log('Subscription:', data.subscription_score);
    console.log('GMP:', data.gmp_score);
    console.log('Financials:', data.financials_score);
    console.log('Listing Predictor:', data.listing_score);
    console.log('Post-Listing:', data.post_listing_score);
    console.log('Risk:', data.risk_score);
    
    return data;
};
