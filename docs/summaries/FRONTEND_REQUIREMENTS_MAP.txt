📝 EXAMPLE API CALL FROM FRONTEND
═══════════════════════════════════════════════════════════════════════════

const analyzeIPO = async (companyName, symbol) => {
    // Show loading state
    setLoading(true);
    
    try {
        const response = await fetch(
            'http://localhost:8000/api/ipo_ultimate/analyze-ultimate',
            {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    company_name: companyName,
                    symbol: symbol,
                    issue_price: 1000
                })
            }
        );
        
        const data = await response.json();
        
        // Display in ALL 4 sections:
        
        // Section 1: Agentic AI Hub
        displayAgenticHub({
            crewai_agents: [
                data.subscription_agent,
                data.gmp_agent,
                data.financials_agent
            ],
            adk_agents: [
                data.listing_agent,
                data.post_listing_agent,
                data.risk_agent
            ],
            context_flow: data.context_flow
        });
        
        // Section 2: IPO Multi-Agent (MAIN)
        displayIPOMultiAgent({
            all_agents: [
                data.subscription_agent,
                data.gmp_agent,
                data.financials_agent,
                data.listing_agent,
                data.post_listing_agent,
                data.risk_agent
            ],
            dynamic_strategy: data.dynamic_strategy,
            final_recommendation: data.final_recommendation,
            overall_score: data.overall_score,
            strengths: data.strengths,
            risks: data.risks,
            tools_used: data.tools_used,
            security_checks: data.security_checks,
            performance: data.performance
        });
        
        // Section 3: Enhanced Analysis
        displayEnhancedAnalysis({
            context_flow: data.context_flow,
            audit_trail: data.audit_trail,
            agent_reasoning: data // Full data
        });
        
        // Section 4: Agent Control Tower
        displayControlTower({
            monitoring_summary: data.monitoring_summary,
            trace_id: data.trace_id,
            audit_trail: data.audit_trail,
            performance: data.performance
        });
        
    } catch (error) {
        console.error('Analysis failed:', error);
        showError(error.message);
    } finally {
        setLoading(false);
    }
};
